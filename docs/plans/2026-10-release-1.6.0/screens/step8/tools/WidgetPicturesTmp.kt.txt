    @Test
    fun pictures() {
        val repo = SettingsRepository(context)
        runBlocking { repo.setLocation(Coordinates(51.5074, -0.1278), "London") }
        val only = args.getString("pic")
        for (scene in scenes()) {
            if (only != null && !scene.name.startsWith(only)) continue
            var shot: Bitmap? = null
            inst.runOnMainSync { shot = draw(scene) }
            File(outDir, "${scene.name}.png").outputStream().use { shot!!.compress(Bitmap.CompressFormat.PNG, 100, it) }
        }
    }

    private fun px(dp: Int) = (dp * density).toInt()

    private fun draw(scene: Scene): Bitmap {
        val root = LinearLayout(context).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(px(16), px(16), px(16), px(16))
            background = GradientDrawable(
                GradientDrawable.Orientation.TL_BR,
                if (scene.dark) intArrayOf(Color.parseColor("#0E1218"), Color.parseColor("#2A3441")) else intArrayOf(Color.parseColor("#AEBBD0"), Color.parseColor("#DCE3EC")),
            )
        }
        val ink = if (scene.dark) Color.parseColor("#C8CED6") else Color.parseColor("#2B3340")
        for (row in scene.rows) {
            val line = LinearLayout(context).apply { orientation = LinearLayout.HORIZONTAL; gravity = Gravity.TOP }
            for (cell in row) {
                val column = LinearLayout(context).apply { orientation = LinearLayout.VERTICAL; setPadding(0, 0, px(16), px(14)) }
                val (view, note) = widget(cell, scene.dark)
                column.addView(TextView(context).apply { text = cell.caption + note; setTextSize(TypedValue.COMPLEX_UNIT_SP, 11f); setTextColor(ink); setPadding(px(2), 0, 0, px(4)) })
                column.addView(FrameLayout(context).apply { addView(view, FrameLayout.LayoutParams(px(cell.w), px(cell.h))) }, LinearLayout.LayoutParams(px(cell.w), px(cell.h)))
                line.addView(column)
            }
            root.addView(line)
        }
        root.measure(View.MeasureSpec.makeMeasureSpec(0, View.MeasureSpec.UNSPECIFIED), View.MeasureSpec.makeMeasureSpec(0, View.MeasureSpec.UNSPECIFIED))
        root.layout(0, 0, root.measuredWidth, root.measuredHeight)
        return Bitmap.createBitmap(root.measuredWidth, root.measuredHeight, Bitmap.Config.ARGB_8888).also { root.draw(Canvas(it)) }
    }

    private fun widget(cell: Cell, dark: Boolean): Pair<View, String> {
        val config = Configuration(context.resources.configuration).apply {
            fontScale = cell.scale
            cell.lang?.let { setLocale(Locale.forLanguageTag(it)) }
        }
        val ctx = context.createConfigurationContext(config)
        val now = LocalDateTime.parse(cell.at).atZone(zone).toInstant()
        val settings = runBlocking { SettingsRepository(context).current() }
        var state = WidgetState.build(settings, now, zone, true).copy(theme = if (dark) ThemeChoice.DARK else ThemeChoice.LIGHT, cityName = cell.city)
        if (cell.noLoc) state = state.copy(hasLocation = false, today = null, next = null, progress = emptyList(), cityName = "")
        WidgetRenderer.stripRamadanWords = cell.ramadanWords
        WidgetRenderer.stripGlyphs = cell.glyphs
        val views: RemoteViews = try {
            when (cell.kind) {
                "strip" -> WidgetRenderer.strip(ctx, state, now)
                "next" -> WidgetRenderer.nextPrayer(ctx, state, now)
                else -> WidgetRenderer.today(ctx, state, now)
            }
        } finally {
            WidgetRenderer.stripRamadanWords = true
            WidgetRenderer.stripGlyphs = true
        }
        val host = AppWidgetHostView(AppLocale.wrap(ctx))
        host.updateAppWidgetSize(Bundle(), listOf(SizeF(cell.w.toFloat(), cell.h.toFloat())))
        host.updateAppWidget(views)
        return host to " [${state.period.name.lowercase()}]"
    }

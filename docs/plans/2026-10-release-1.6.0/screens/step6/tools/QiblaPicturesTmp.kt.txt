class QiblaPicturesTmp {
    @get:Rule val compose = createComposeRule()

    private val args = InstrumentationRegistry.getArguments()

    @Test
    fun picture() {
        val name = args.getString("pic") ?: "pic"
        val heading = args.getString("heading")?.toDouble()
        val dark = args.getString("dark") == "1"
        val scale = args.getString("scale")?.toFloat() ?: 1f
        val narrow = args.getString("narrow") == "1"
        val noLocation = args.getString("noloc") == "1"
        val open = args.getString("open") == "1"
        qiblaCentreInstruction = args.getString("centre") == "1"
        qiblaSandBezel = args.getString("sand") == "1"
        qiblaShowArc = args.getString("arc") == "1"
        qiblaDarkBezelBright = args.getString("bright") == "1"
        val accuracy = when (args.getString("acc")) {
            "low" -> CompassAccuracy.LOW
            "none" -> CompassAccuracy.UNAVAILABLE
            else -> CompassAccuracy.HIGH
        }
        val state = UiState(
            loading = false,
            settings = AppSettings(cityName = "London"),
            qiblaBearing = if (noLocation) null else 118.0,
            qiblaDistanceKm = if (noLocation) null else 4794.0,
            compassHeading = if (accuracy == CompassAccuracy.UNAVAILABLE) null else heading,
            compassAccuracy = accuracy,
        )
        compose.setContent {
            val density = LocalDensity.current
            CompositionLocalProvider(LocalDensity provides Density(density.density, fontScale = scale)) {
                SajdaTimeTheme(if (dark) ThemeChoice.DARK else ThemeChoice.LIGHT) {
                    androidx.compose.material3.Surface(Modifier.fillMaxSize(), color = MaterialTheme.colorScheme.background) { Box(contentAlignment = Alignment.TopCenter) {
                        Box(if (narrow) Modifier.width(320.dp).statusBarsPadding() else Modifier.fillMaxSize().statusBarsPadding()) {
                            QiblaScreen(state, onUseGps = {}, onSearchCity = {})
                            GlassTabBar(Destination.QIBLA, onSelect = {}, modifier = Modifier.align(Alignment.BottomCenter))
                        }
                    } }
                }
            }
        }
        if (open) compose.onNode(hasText(InstrumentationRegistry.getInstrumentation().targetContext.getString(R.string.notice_show_details)) and hasClickAction()).performClick()
        compose.waitForIdle()
        Thread.sleep(1500)
        val shot: Bitmap = InstrumentationRegistry.getInstrumentation().uiAutomation.takeScreenshot()
        val dir = InstrumentationRegistry.getInstrumentation().targetContext.getExternalFilesDir(null)!!
        File(dir, "$name.png").outputStream().use { shot.compress(Bitmap.CompressFormat.PNG, 100, it) }
    }
}
